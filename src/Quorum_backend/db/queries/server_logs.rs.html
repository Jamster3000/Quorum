1//! File to handle logging server events
2//!
3//! This module defines functions to log different types of server events (startup, shutdown, errors) into the database.
4//! It uses a caching mechanism to minimize database queries for event type IDs, ensuring efficient logging.
5
6use crate::db::DB;
7use std::error::Error;
8use std::sync::OnceLock;
9use surrealdb_types::{RecordId, SurrealValue};
10
11static STARTUP_ID: OnceLock<RecordId> = OnceLock::new();
12static SHUTDOWN_ID: OnceLock<RecordId> = OnceLock::new();
13static ERROR_ID: OnceLock<RecordId> = OnceLock::new();
14
15#[derive(Debug, SurrealValue)]
16struct EventTypeRecord {
17    id: RecordId,
18}
19
20/// Gets the event type ID for the given event name, creating a new event type if it doesn't already exist.
21///
22/// This function performs a database query to check if the event type already exists. If it does, it returns the existing ID.
23/// If it doesn't exist, it creates a new event type and returns the new ID.
24///
25/// # Arguments
26/// * `db` - A reference to the database connection.
27/// * `event_name` - The name of the event type to get or create.
28///
29/// # Returns
30/// * `Ok(RecordId)` - The ID of the event type, either existing or newly created.
31/// * `Err(Box<dyn Error>)` - An error if the database query fails or if creating a new event type fails.
32///
33/// # Errors
34/// * "Failed to create event type" - If the database query to create a new event type does not return a valid record, indicating that the creation failed.
35async fn get_or_create_event_type(db: &DB, event_name: String) -> Result<RecordId, Box<dyn Error>> {
36    //Check if the event type already exists by trying to get it.
37    let mut response = db
38        .query("SELECT id FROM server_log_event_types WHERE name = $name")
39        .bind(("name", event_name.clone()))
40        .await?;
41
42    let result: Option<EventTypeRecord> = response.take(0)?;
43
44    //If event type exists already in `server_log_event_types` return its ID
45    if let Some(record) = result {
46        Ok(record.id)
47    } else {
48        //event type doesn't exist, create it then return its ID
49        let mut create_response = db
50            .query("CREATE server_log_event_types SET name = $name RETURN id")
51            .bind(("name", event_name))
52            .await?;
53
54        let created: Option<EventTypeRecord> = create_response.take(0)?;
55        Ok(created.ok_or("Failed to create event type")?.id)
56    }
57}

59/// Gets the cached event type ID for the given event name, using a OnceLock to cache the ID after the first retrieval.
60///
61/// This function first checks the provided OnceLock cache for the event type ID. If the ID is already cached, it returns it immediately.
62/// If the ID is not cached, it calls `get_or_create_event_type` to retrieve or create the event type ID from the database, caches it in the OnceLock, and then returns it.
63///
64/// # Arguments
65/// * `db` - A reference to the database connection.
66/// * `cache` - A reference to a OnceLock that will cache the event type ID after the first retrieval.
67/// * `event_name` - The name of the event type to get or create.
68///
69/// # Returns
70/// * `Ok(RecordId)` - The ID of the event type, either retrieved from the cache or obtained from the database.
71/// * `Err(Box<dyn Error>)` - An error if the database query fails or if creating a new event type fails.
72///
73/// # Errors
74/// * "Failed to create event type" - If the database query to create a new event type does not return a valid record, indicating that the creation failed.
75async fn get_cached_event_type(
76    db: &DB,
77    cache: &OnceLock<RecordId>,
78    event_name: &str,
79) -> Result<RecordId, Box<dyn Error>> {
80    if let Some(id) = cache.get() {
81        return Ok(id.clone());
82    }
83
84    let id = get_or_create_event_type(db, event_name.to_string()).await?;
85    let _ = cache.set(id.clone());
86    Ok(id)
87}

89/// Logs a startup event to the database with the given duration in milliseconds.
90///
91/// This function retrieves the event type ID for "startup" using the caching mechanism and then creates a new log entry in the `server_logs` table with the event type ID and duration.
92///
93/// # Arguments
94/// * `db` - A reference to the database connection.
95/// * `duration_ms` - The duration of the startup event in milliseconds.
96///
97/// # Returns
98/// * `Ok(())` - If the log entry was successfully created.
99/// * `Err(Box<dyn Error>)` - An error if the database query fails or if retrieving the event type ID fails.
100///
101/// # Errors
102/// * "Failed to create event type" - If the database query to create a new event type does not return a valid record, indicating that the creation failed.
103///
104/// # Example
105/// ```
106/// use crate::db::DB;
107/// use crate::db::queries::server_logs;
108/// async fn example_log_startup(db: &DB) {
109///     let duration_ms = 1500;
110///     match server_logs::log_startup(db, duration_ms).await {
111///         Ok(()) => println!("Startup event logged successfully"),
112///         Err(e) => eprintln!("Error logging startup event: {}", e),
113///     }
114/// }
115/// ```
116pub async fn log_startup(db: &DB, duration_ms: i64) -> Result<(), Box<dyn Error>> {
117    let event_type_id = get_cached_event_type(db, &STARTUP_ID, "startup").await?;
118
119    db.query("CREATE server_logs SET event_type_id = $event_type_id, duration_ms = $duration_ms")
120        .bind(("event_type_id", event_type_id))
121        .bind(("duration_ms", duration_ms))
122        .await?;
123
124    Ok(())
125}

165/// Logs an error event to the database with the given message and error code.
166///
167/// This function retrieves the event type ID for "error" using the caching mechanism and then creates a new log entry in the `server_logs` table with the event type ID, message, and error code.
168///
169/// # Arguments
170/// * `db` - A reference to the database connection.
171/// * `message` - A string describing the error message to log.
172/// * `error_code` - A numeric code representing the error to log.
173///
174/// # Returns
175/// * `Ok(())` - If the log entry was successfully created.
176/// * `Err(Box<dyn Error>)` - An error if the database query fails or if retrieving the event type ID fails.
177///
178/// # Errors
179/// * "Failed to create event type" - If the database query to create a new event type does not return a valid record, indicating that the creation failed.
180///
181/// # Example
182/// ```
183/// use crate::db::DB;
184/// use crate::db::queries::server_logs;
185/// async fn example_log_error(db: &DB) {
186///     let message = "An unexpected error occurred".to_string();
187///     let error_code = 500;
188///     match server_logs::log_error(db, message, error_code).await {
189///         Ok(()) => println!("Error event logged successfully"),
190///         Err(e) => eprintln!("Error logging error event: {}", e),
191///     }
192/// }
193/// ```
194pub async fn log_error(db: &DB, message: String, error_code: u32) -> Result<(), Box<dyn Error>> {
195    let event_type_id = get_cached_event_type(db, &ERROR_ID, "error").await?;
196
197    db.query(
198        "CREATE server_logs SET event_type_id = $event_type_id, message = $message, error_code = $error_code",
199    )
200    .bind(("event_type_id", event_type_id))
201    .bind(("message", message))
202    .bind(("error_code", error_code))
203    .await?;
204
205    Ok(())
