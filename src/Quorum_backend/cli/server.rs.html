8/// Displays server logs
9///
10/// Shows the server logs from the `server_logs` database table.
11/// Has option to show the last X days of logs.
12/// By default, shows 100 of the most recent logs.
13///
14/// # Arguments
15/// * `db` - A reference to the database connection.
16/// * `raw` - The raw input string, which may contain the number of days to filter logs by.
17///
18/// # Example
19/// ```rust
20/// logs(&db, "").await; // Shows the latest 100 logs
21/// logs(&db, "7").await; // Shows logs from the last 7 days
22/// logs(&db, "invalid").await; // Shows usage message
23/// ```
24pub async fn logs(db: &DB, raw: &str) {
25    let days: Option<u32> = raw.trim().parse().ok();
26
27    if !raw.trim().is_empty() && days.is_none() {
28        println!("{}", "Usage: server:logs OR server:logs <days>".red());
29        return;
30    }
31
32    match crate::db::queries::logs::get_server_logs(db, days).await {
33        Err(e) => println!("{}", format!("  Failed to get logs: {}", e).red()),
34        Ok(entries) => {
35            println!();
36            println!("{}", "  Server Logs".cyan().bold());
37            if let Some(d) = days {
38                println!("  {}", format!("Last {} days", d).dimmed());
39            } else {
40                println!("  {}", "Latest 100 entries".dimmed());
41            }
42            println!(
43                "{}",
44                "  ─────────────────────────────────────────────────────".dimmed()
45            );
46
47            if entries.is_empty() {
48                println!("{}", "  No log entries found.".dimmed());
49            } else {
50                for e in &entries {
51                    let mut parts = vec![
52                        e.timestamp.dimmed().to_string(),
53                        e.event_type.cyan().to_string(),
54                    ];
55                    if let Some(ms) = e.duration_ms {
56                        parts.push(format!("{}ms", ms).dimmed().to_string());
57                    }
58                    if let Some(msg) = &e.message {
59                        parts.push(msg.yellow().to_string());
60                    }
61                    if let Some(code) = e.error_code {
62                        parts.push(format!("code:{}", code).red().to_string());
63                    }
64                    println!("  {}", parts.join("  ·  "));
65                }
66            }
67            println!();
68        }
69    }
70}

136/// Displays server status
137///
138/// Shows the server uptime, listening address, and testing mode status.
139///
140/// # Arguments
141/// * `server_start` - The instant when the server started, used for uptime calculations.
142///
143/// # Example
144/// ```rust
145/// status(server_start).await; // Displays the server status
146/// ```
147pub async fn status(server_start: Instant) {
148    let elapsed = server_start.elapsed();
149
150    let total_secs = elapsed.as_secs();
151    let days = total_secs / 86400;
152    let hours = (total_secs % 86400) / 3600;
153    let mins = (total_secs % 3600) / 60;
154    let secs = total_secs % 60;
155
156    let uptime = if days > 0 {
157        format!("{}d {}h {}m {}s", days, hours, mins, secs)
158    } else if hours > 0 {
159        format!("{}h {}m {}s", hours, mins, secs)
160    } else if mins > 0 {
161        format!("{}m {}s", mins, secs)
162    } else {
163        format!("{}s", secs)
164    };
165
166    let config = crate::utility::config::Config::get();
167
168    println!();
169    println!("{}", "  Server Status".cyan().bold());
170    println!("{}", "  ─────────────────────────────────".dimmed());
171    println!("  {:<20} {}", "Uptime:".white(), uptime.green());
172    println!(
173        "  {:<20} {}",
174        "Listening on:".white(),
175        format!("{}:{}", config.server_host, config.server_port).green()
176    );
177    println!(
178        "  {:<20} {}",
179        "Testing mode:".white(),
180        if config.enable_testing {
181            "enabled".yellow().to_string()
182        } else {
183            "disabled".dimmed().to_string()
184        }
185    );
186    println!();
187}

189/// Handles user signup
190///
191/// Allows anyone with access to the server to be able to signup and create an account.
192/// When this is called it prompts for username, email (optional) and password one by one, where password is typed hidden.
193///
194/// # Arguments
195/// * `db` - A reference to the database connection.
196///
197/// # Example
198/// ```rust
199/// signup(&db).await; // Prompts for username, password, and optional email to create a new user
200/// ```
201pub async fn signup(db: &DB) {
202    let username = prompt("  Username: ");
203    let password = prompt_password("  Password: ");
204    let email = prompt("  Email (optional, leave blank): ");
205
206    let email = if email.is_empty() { None } else { Some(email) };
207
208    match crate::db::queries::auth::signup_user(db, &username, email.as_deref(), &password, None)
209        .await
210    {
211        Ok(user) => println!(
212            "{}",
213            format!(
214                "  Created user: {} ({})",
215                user.username,
216                format!("{:?}", user.id.key)
217            )
218            .green()
219        ),
220        Err(e) => println!("{}", format!("  Failed to create user: {}", e).red()),
221    }
222}

224/// Handles user login
225///
226/// Prompts for username and password, and verifies the credentials against the database.
227/// When this is called it prompts for username and password one by one, where password is typed hidden.
228///
229/// # Arguments
230/// * `db` - A reference to the database connection.
231/// * `session` - A reference to the admin session, used to track login state.
232///
233/// # Example
234/// ```rust
235/// login(&db, &session).await; // Prompts for username and password to log in
236/// ```
237pub async fn login(db: &DB, session: &Arc<Mutex<AdminSession>>) {
238    {
239        let sess = session.lock().unwrap();
240        if sess.is_valid() {
241            println!(
242                "{}",
243                format!(
244                    "Already logged in as {}. Run server:logout first.",
245                    sess.username.as_deref().unwrap_or("unknown")
246                )
247                .yellow()
248            );
249            return;
250        }
251    }
252
253    let username = prompt("  Username: ");
254    let password = prompt_password("  Password: ");
255
256    match crate::db::queries::auth::verify_user_credentials(db, &username, &password).await {
257        Ok(user) => {
258            let mut sess = session.lock().unwrap();
259            sess.login(user.username.clone(), user.is_admin);
260            println!(
261                "{}",
262                format!("  Logged in as {}.", user.username).green().bold()
263            );
264            println!(
265                "{}",
266                format!(
267                    "  Session will expire after {} minutes of inactivity.",
268                    crate::cli::SESSION_TIMEOUT_MINS
269                )
270                .dimmed()
271            );
272        }
273        Err(_) => {
274            println!("{}", "  Invalid username or password.".red());
275        }
276    }
277
278    println!();
279}

281/// Handles user logout
282///
283/// Logs out the currently logged-in user, if any.
284///
285/// # Arguments
286/// * `session` - A reference to the admin session, used to track login state.
287///
288/// # Example
289/// ```rust
290/// logout(&session); // Logs out the currently logged-in user
291/// ```
292pub fn logout(session: &Arc<Mutex<AdminSession>>) {
293    let mut sess = session.lock().unwrap();
294    if !sess.logged_in {
295        println!("{}", "Not currently logged in.".yellow());
296        return;
297    }
298    let username = sess.username.clone().unwrap_or_default();
299    sess.logout();
300    println!("{}", format!("Logged out: {}.", username).green());
301}

303/// Handles server shutdown
304///
305/// Logs the shutdown event to the database and signals the server to shut down gracefully.
306///
307/// # Arguments
308/// * `db` - A reference to the database connection.
309/// * `server_start` - The instant when the server started, used for uptime calculations.
310/// * `shutdown_tx` - A watch channel sender to signal server shutdown.
311///
312/// # Example
313/// ```rust
314/// shutdown(&db, server_start, &shutdown_tx).await; // Logs shutdown and signals server to shut down
315/// ```
316pub async fn shutdown(db: &DB, server_start: Instant, shutdown_tx: &watch::Sender<bool>) {
317    println!("{}", "\nShutting down gracefully...".yellow().bold());
318
319    let uptime_ms = server_start.elapsed().as_millis() as i64;
320    let _ = crate::db::queries::server_logs::log_shutdown(db, uptime_ms).await;
321
322    let _ = shutdown_tx.send(true);
323    std::process::exit(0);
324}

326/// Handles making a user an admin
327///
328/// This uses a username as reference to mark a user as an admin.
329///
330/// # Arguments
331/// * `db` - A reference to the database connection.
332/// * `username` - The username of the user to be made an admin.

346                sess.update_is_admin(true);
347            }
348        }
349        Err(e) => println!("{}", format!("  Failed: {}", e).red()),
350    }

368/// Prompts the user for input with a given label and returns the trimmed input as a String.
369///
370/// # Arguments
371/// * `label` - The label to display before the input prompt.
372///
373/// # returns
374/// A String containing the user's input, trimmed of whitespace.
375fn prompt(label: &str) -> String {
376    print!("{}", label.white());
377    std::io::Write::flush(&mut std::io::stdout()).ok();
378
379    let mut input = String::new();
380    std::io::stdin().read_line(&mut input).ok();
381    input.trim().to_string()
382}
