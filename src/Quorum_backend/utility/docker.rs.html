1//! Utility functions for managing Docker containers
2//! If the docker container for SurrealDB and minio isn't running the
3//! `ensure_containers_running` funtion will check and execute `docker compose up -d`
4//! to start the containers.
5
6use std::error::Error;
7use std::process::Command;
8
9/// ensure_containers_running checks if the required Docker
10/// containers are running and starts them if they are not.
11///
12/// # returns
13/// Returns Ok(()) if the containers are running or were started successfully.
14///
15/// # Errors
16/// Returns an error if the docker-compose.yml file is not found,
17/// if the docker command fails to execute, or if the containers fail to start.
18///
19/// # Examples
20/// ```
21/// use server::utility::docker::ensure_containers_running;
22/// tokio_test::block_on(async {
23///     ensure_containers_running().await.unwrap();
24/// });
25/// ```
26pub async fn ensure_containers_running() -> Result<(), Box<dyn Error>> {
27    let docker_dir = std::path::PathBuf::from(env!("CARGO_MANIFEST_DIR"))
28        .parent()
29        .ok_or("Failed to determine project root")?
30        .join("docker");
31
32    if !docker_dir.join("docker-compose.yml").exists() {
33        return Err("docker-compose.yml not found in docker directory".into());
34    }
35
36    let output = Command::new("docker")
37        .args(["compose", "up", "-d"])
38        .current_dir(&docker_dir)
39        .output()
40        .map_err(|_| "Failed to execute docker compose command")?;
41
42    if !output.status.success() {
43        let stderr = String::from_utf8_lossy(&output.stderr);
44        return Err(format!("Failed to start Docker containers: {}", stderr.trim()).into());
45    }
46
47    tokio::time::sleep(tokio::time::Duration::from_millis(500)).await;
48
49    Ok(())
50}
