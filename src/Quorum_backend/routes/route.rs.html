25/// Creates the router with all the defined routes and their handlers.
26/// Includes request limiting to avoid getting too many requests at once.
27///
28/// # Arguments
29/// * `db` - The database connection pool.
30///
31/// # Returns
32/// A configured `Router` instance with all routes and their handlers.
33pub fn create_router(db: DB) -> Router {
34    let cors = CorsLayer::new()
35        .allow_origin(Any) // replace with more specific origin for added safety
36        .allow_methods(Any)
37        .allow_headers(Any);
38
39    let security_headers = ServiceBuilder::new()
40        .layer(SetResponseHeaderLayer::if_not_present(
41            HeaderName::from_static("x-frame-options"),
42            HeaderValue::from_static("DENY"),
43        ))
44        .layer(SetResponseHeaderLayer::if_not_present(
45            HeaderName::from_static("strict-transport-security"),
46            HeaderValue::from_static("max-age=31536000; includeSubDomains; preload"),
47        ))
48        .layer(SetResponseHeaderLayer::if_not_present(
49            HeaderName::from_static("x-content-type-options"),
50            HeaderValue::from_static("nosniff"),
51        ));
52
53    Router::new()
54        .route("/", get(|| async { "Axum server is running" }))
55        .route("/health", get(health))
56        .route("/echo", post(echo))
57        .merge(auth_routes())
58        .layer(cors) //can be replaced with .allow_origin("http://localhost:xxxx".parse::<HeaderValue>().unwrap()) for more specific origin safety
59        .layer(security_headers)
60        .with_state(db)
61}

63/// Builds the auth route group with an appropriate rate limiter.
64///
65/// In testing mode, a more permissive rate limit is applied so the test suite
66/// can run without hitting the default limits.
67fn auth_routes() -> Router<DB> {
68    let config = Config::get();
69
70    let (per_second, burst_size) = if config.enable_testing {
71        (config.testing_per_second, config.testing_burst_size)
72    } else {
73        (config.default_per_second, config.default_burst_size)
74    };
75
76    let governor_conf = Arc::new(
77        GovernorConfigBuilder::default()
78            .per_second(per_second)
79            .burst_size(burst_size)
80            .finish()
81            .unwrap(),
82    );
83
84    // Periodically evict expired rate limit entries to keep memory bounded
85    let limiter = governor_conf.limiter().clone();
86    std::thread::spawn(move || {
87        loop {
88            std::thread::sleep(Duration::from_secs(60));
89            limiter.retain_recent();
90        }
91    });
92
93    Router::new()
94        .route("/auth/signup", post(signup))
95        .route("/auth/login", post(login))
96        .route("/auth/delete", post(delete_account))
97        .route("/auth/me", post(get_user_data))
98        .route("/auth/refresh", post(refresh_token))
99        .route("/auth/logout", post(logout))
100        .route("/auth/updateuserprofile", post(update_user_profile))
101        .layer(GovernorLayer::new(governor_conf))
102}
