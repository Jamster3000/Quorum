27/// Verifies user credentials (username/email and password) against the database
28///
29// # Arguments
30/// * `db` - A reference to the database connection
31/// * `username_or_email` - The username or email provided by the user
32/// * `password` - The password provided by the user
33///
34/// # Returns
35/// * `Ok(User)` - The User struct if the credentials are valid
36/// * `Err((StatusCode, String))` - A tuple containing the HTTP status code and an error message if the credentials are invalid
37async fn verify_user_credentials(
38    db: &DB,
39    username_or_email: &str,
40    password: &str,
41) -> Result<User, (StatusCode, String)> {
42    auth::verify_user_credentials(db, username_or_email, password)
43        .await
44        .map_err(|_| {
45            (
46                StatusCode::UNAUTHORIZED,
47                "Invalid username/email or password".to_string(),
48            )
49        })
50}

52/// Extracts the user ID as a string from the User struct
53///
54/// # Arguments
55/// * `user` - The User struct from which to extract the ID
56///
57/// # Returns
58/// * `Ok(String)` - The user ID as a string if extraction is successful
59/// * `Err(String)` - An error message if the ID format is invalid
60fn extract_user_id(user: &User) -> Result<String, String> {
61    match &user.id.key {
62        surrealdb_types::RecordIdKey::String(s) => Ok(s.to_string()),
63        surrealdb_types::RecordIdKey::Number(n) => Ok(n.to_string()),
64        _ => Err("Invalid user ID format".to_string()),
65    }
66}

68/// Handles user signup requests
69///
70/// # Arguments
71/// * `State(db)` - Shared state containing the database connection and JWT configuration
72/// * `Json(payload)` - The signup request payload containing username, email, and password
73///
74/// # Returns
75/// * `(StatusCode, Json<AuthTokenResponse>)` - The HTTP status code and JSON response containing the authentication token or error message
76///
77/// # Error
78/// * `StatusCode::INTERNAL_SERVER_ERROR` - If there is an error processing the password, creating the user, generating tokens, or storing the refresh token
79/// * `StatusCode::CONFLICT` - If the username already exists
80/// * `StatusCode::BAD_REQUEST` - If there is a user ID mismatch during account deletion
81///
82/// # Example
83/// ```rust
84/// let payload = SignupRequest {
85///     username: "newuser".to_string(),
86///     email: Some("newuser@example.com".to_string()),
87///     password: "password123".to_string(),
88/// };
89/// ```
90pub async fn signup(
91    State(db): State<DB>,
92    Json(payload): Json<SignupRequest>,
93) -> (StatusCode, Json<AuthTokenResponse>) {
94    let mut _plain_backup_codes = None;
95    let mut hash_salt_backup_codes = None;
96
97    if payload.email.is_none() {
98        let backup_codes = Some(generate_backup_codes());
99
100        _plain_backup_codes = Some(
101            backup_codes
102                .as_ref()
103                .unwrap()
104                .iter()
105                .filter_map(|code| code.plain.clone())
106                .collect::<Vec<String>>(),
107        );
108        hash_salt_backup_codes = Some(
109            backup_codes
110                .as_ref()
111                .unwrap()
112                .iter()
113                .map(|code| (code.hash.clone(), code.salt.clone()))
114                .collect::<Vec<(String, String)>>(),
115        );
116    }
117
118    let signup_result = auth::signup_user(
119        &db,
120        &payload.username,
121        payload.email.as_deref(),
122        &payload.password,
123        hash_salt_backup_codes.clone(),
124    )
125    .await;
126
127    if let Err(e) = signup_result {
128        let message = e.to_string();
129        let formatted = format!("Failed to create user: {}", message);
130
131        let status = if message.contains("Invalid password length")
132            || message.contains("idx_username")
133            || message.contains("idx_email")
134        {
135            StatusCode::BAD_REQUEST
136        } else if message.contains("idx_username") || message.contains("UNIQUE") {
137            StatusCode::CONFLICT
138        } else {
139            StatusCode::INTERNAL_SERVER_ERROR
140        };
141
142        let _ = log_audit_event(
143            &db,
144            AuditEvent {
145                log_type: "signup_failed".to_string(),
146                action: Some(message.clone()),
147                ..Default::default()
148            },
149        )
150        .await;
151
152        return (
153            status,
154            Json(AuthTokenResponse {
155                success: false,
156                user: None,
157                tokens: None,
158                message: formatted,
159            }),
160        );
161    }
162
163    let user = signup_result.unwrap();
164
165    let user_id = match extract_user_id(&user) {
166        Ok(id) => id,
167        Err(e) => {
168            return (
169                StatusCode::INTERNAL_SERVER_ERROR,
170                Json(AuthTokenResponse {
171                    success: false,
172                    user: None,
173                    tokens: None,
174                    message: e,
175                }),
176            );
177        }
178    };
179
180    let access_token = match crate::utility::jwt::generate_access_token(&user_id, &user.username) {
181        Ok(token) => token,
182        Err(_) => {
183            return (
184                StatusCode::INTERNAL_SERVER_ERROR,
185                Json(AuthTokenResponse {
186                    success: false,
187                    user: None,
188                    tokens: None,
189                    message: "Failed to generate access token".to_string(),
190                }),
191            );
192        }
193    };
194
195    let refresh_token = match crate::utility::jwt::generate_refresh_token(&user_id, &user.username)
196    {
197        Ok(token) => token,
198        Err(_) => {
199            return (
200                StatusCode::INTERNAL_SERVER_ERROR,
201                Json(AuthTokenResponse {
202                    success: false,
203                    user: None,
204                    tokens: None,
205                    message: "Failed to generate refresh token".to_string(),
206                }),
207            );
208        }
209    };
210
211    let config = quorum_core::utility::config::Config::get();
212    let expires_at = chrono::Utc::now().timestamp() + (config.jwt_refresh_days * 86400);
213    if auth::store_refresh_token(&db, &user_id, &refresh_token, expires_at)
214        .await
215        .is_err()
216    {
217        return (
218            StatusCode::INTERNAL_SERVER_ERROR,
219            Json(AuthTokenResponse {
220                success: false,
221                user: None,
222                tokens: None,
223                message: "Failed to store refresh token".to_string(),
224            }),
225        );
226    }
227
228    let _ = log_audit_event(
229        &db,
230        AuditEvent {
231            log_type: "signup_success".to_string(),
232            target_type_table: Some("users".to_string()),
233            target_type_table_id: Some(user_id.clone()),
234            user_id: Some(user_id.clone()),
235            ..Default::default()
236        },
237    )
238    .await;
239
240    (
241        StatusCode::CREATED,
242        Json(AuthTokenResponse {
243            success: true,
244            user: Some(user.to_response()),
245            tokens: Some(TokenResponse {
246                access_token,
247                refresh_token,
248                expires_in: quorum_core::utility::config::Config::get().jwt_access_minutes * 60,
249            }),
250            message: "User created successfully".to_string(),
251        }),
252    )
253}

480/// Handles user data retrieval requests
481///
482/// # Arguments
483/// * `State(db)` - Shared state containing the database connection
484/// * `Json(payload)` - The user data retrieval request payload containing username/email, password, and requested fields
485///
486/// # Returns
487/// * `(StatusCode, Json<UserDataResponse>)` - The HTTP status code and JSON response containing the requested user data or an error message
488///
489/// # Example
490/// ```rust
491/// let payload = GetUserDataRequest {
492///     username_or_email: "john_doe",
493///     password: "secret_password",
494///     user_id: "user_id",
495///     fields: vec!["id".to_string(), "username".to_string()],
496/// };
497/// ```
498pub async fn get_user_data(
499    State(db): State<DB>,
500    Json(payload): Json<GetUserDataRequest>,
501) -> (StatusCode, Json<UserDataResponse>) {
502    let user =
503        match verify_user_credentials(&db, &payload.username_or_email, &payload.password).await {
504            Ok(user) => user,
505            Err((status, message)) => {
506                let _ = log_audit_event(
507                    &db,
508                    AuditEvent {
509                        log_type: "get_user_data_failed".to_string(),
510                        action: Some(message.clone()),
511                        target_type_table: Some("users".to_string()),
512                        target_type_table_id: Some(payload.username_or_email.clone()),
513                        user_id: Some(payload.user_id),
514                        ..Default::default()
515                    },
516                )
517                .await;
518
519                return (
520                    status,
521                    Json(UserDataResponse {
522                        success: false,
523                        data: None,
524                        message,
525                    }),
526                );
527            }
528        };
529
530    let mut user_data = serde_json::Map::new();
531
532    for field in &payload.fields {
533        match field.as_str() {
534            "id" => {
535                user_data.insert(
536                    "id".to_string(),
537                    serde_json::Value::String(format!("{:?}", user.id.key)),
538                );
539            }
540            "username" => {
541                user_data.insert(
542                    "username".to_string(),
543                    serde_json::Value::String(user.username.clone()),
544                );
545            }
546            "email" => {
547                if let Some(email) = &user.email {
548                    user_data.insert(
549                        "email".to_string(),
550                        serde_json::Value::String(email.clone()),
551                    );
552                }
553            }
554            "created_at" => {
555                user_data.insert(
556                    "created_at".to_string(),
557                    serde_json::Value::String(user.created_at.to_string()),
558                );
559            }
560            _ => {}
561        }
562    }
563
564    let _ = log_audit_event(
565        &db,
566        AuditEvent {
567            log_type: "get_user_data_success".to_string(),
568            target_type_table: Some("users".to_string()),
569            target_type_table_id: Some(payload.user_id.clone()),
570            user_id: Some(payload.user_id),
571            ..Default::default()
572        },
573    )
574    .await;
575
576    (
577        StatusCode::OK,
578        Json(UserDataResponse {
579            success: true,
580            data: Some(user_data),
581            message: "User data retrieved successfully".to_string(),
582        }),
583    )
584}

586/// Handles token refresh requests
587/// This endpoint allows clients to obtain a new access token using a valid refresh token. It verifies the refresh token, checks its validity in the database, and generates a new access token if everything is valid. The refresh token is not rotated in this implementation, but it can be easily modified to do so if desired.
588///
589/// # Arguments
590/// * `State(_db)` - Shared state containing the database connection (not used but required by Axum)
591/// * `Json(payload)` - The refresh token request payload containing the refresh token
592///
593/// # Returns
594/// * `(StatusCode, Json<AuthTokenResponse>)` - The HTTP status code and JSON response containing the new access token or an error message
595///
596/// # Error
597/// * `StatusCode::UNAUTHORIZED` - If the refresh token is invalid, expired, of the wrong type, or not found in the database
598/// * `StatusCode::INTERNAL_SERVER_ERROR` - If there is an error generating the new access token
599///
600/// # Example
601/// ```rust
602/// let payload = RefreshTokenRequest {
603///     refresh_token: "valid_refresh_token".to_string(),
604/// };
605/// ```
606pub async fn refresh_token(
607    State(db): State<DB>,
608    Json(payload): Json<RefreshTokenRequest>,
609) -> (StatusCode, Json<AuthTokenResponse>) {
610    // Verify token
611    let claims = match crate::utility::jwt::verify_token(&payload.refresh_token) {
612        Ok(claims) => {
613            if claims.token_type != "refresh" {
614                return (
615                    StatusCode::UNAUTHORIZED,
616                    Json(AuthTokenResponse {
617                        success: false,
618                        user: None,
619                        tokens: None,
620                        message: "Invalid token type".to_string(),
621                    }),
622                );
623            }
624            claims
625        }
626        Err(_) => {
627            let user_id = payload.user_id.clone();
628
629            let _ = log_audit_event(
630                &db,
631                AuditEvent {
632                    log_type: "refresh_token_failed".to_string(),
633                    action: Some("Invalid or expired refresh token".to_string()),
634                    target_type_table_id: Some(user_id.clone()),
635                    user_id: Some(user_id),
636                    ..Default::default()
637                },
638            )
639            .await;
640
641            return (
642                StatusCode::UNAUTHORIZED,
643                Json(AuthTokenResponse {
644                    success: false,
645                    user: None,
646                    tokens: None,
647                    message: "Invalid or expired refresh token".to_string(),
648                }),
649            );
650        }
651    };
652
653    //Generate access token
654    let access_token =
655        match crate::utility::jwt::generate_access_token(&claims.sub, &claims.username) {
656            Ok(token) => token,
657            Err(_) => {
658                return (
659                    StatusCode::INTERNAL_SERVER_ERROR,
660                    Json(AuthTokenResponse {
661                        success: false,
662                        user: None,
663                        tokens: None,
664                        message: "Failed to generate access token".to_string(),
665                    }),
666                );
667            }
668        };
669
670    let _ = log_audit_event(
671        &db,
672        AuditEvent {
673            log_type: "refresh_token_success".to_string(),
674            target_type_table: Some("users".to_string()),
675            target_type_table_id: Some(claims.sub.clone()),
676            user_id: Some(claims.sub.clone()),
677            ..Default::default()
678        },
679    )
680    .await;
681
682    (
683        StatusCode::OK,
684        Json(AuthTokenResponse {
685            success: true,
686            user: None,
687            tokens: Some(TokenResponse {
688                access_token,
689                refresh_token: payload.refresh_token,
690                expires_in: Config::get().jwt_access_minutes * 60,
691            }),
692            message: "Token refreshed successfully".to_string(),
693        }),
694    )
695}

697/// Handles user logout requests
698/// This endpoint allows clients to log out by revoking the provided refresh token. It removes the refresh token from the database,
699/// effectively invalidating it and preventing any further use for obtaining new access tokens.
700///
701/// # Arguments
702/// * `State(db)` - Shared state containing the database connection and JWT configuration (JWT config is not used in this handler)
703/// * `Json(payload)` - The logout request payload containing the refresh token to be revoked
704///
705/// # Returns
706/// * `(StatusCode, Json<AuthResponse>)` - The HTTP status code and JSON response indicating success or failure of the logout operation
707///
708/// # Error
709/// * `StatusCode::INTERNAL_SERVER_ERROR` - If there is an error revoking the refresh token
710///
711/// # Example
712/// ```rust
713/// let payload = RefreshTokenRequest {
714///     refresh_token: "valid_refresh_token".to_string(),
715/// };
716/// ```
717pub async fn logout(
718    State(db): State<DB>,
719    Json(payload): Json<RefreshTokenRequest>,
720) -> (StatusCode, Json<AuthTokenResponse>) {
721    if crate::utility::jwt::verify_token(&payload.refresh_token).is_err() {
722        let user_id = payload.user_id.clone();
723
724        let _ = log_audit_event(
725            &db,
726            AuditEvent {
727                log_type: "logout_failed".to_string(),
728                action: Some("Invalid or expired refresh token".to_string()),
729                user_id: Some(user_id),
730                ..Default::default()
731            },
732        )
733        .await;
734
735        return (
736            StatusCode::UNAUTHORIZED,
737            Json(AuthTokenResponse {
738                success: false,
739                user: None,
740                tokens: None,
741                message: "Invalid or expired refresh token".to_string(),
742            }),
743        );
744    }
745
746    if auth::revoke_refresh_token(&db, &payload.refresh_token)
747        .await
748        .is_err()
749    {
750        let user_id = payload.user_id.clone();
751
752        let _ = log_audit_event(
753            &db,
754            AuditEvent {
755                log_type: "logout_failed".to_string(),
756                action: Some("Failed to revoke refresh token".to_string()),
757                user_id: Some(user_id),
758                ..Default::default()
759            },
760        )
761        .await;
762
763        return (
764            StatusCode::INTERNAL_SERVER_ERROR,
765            Json(AuthTokenResponse {
766                success: false,
767                user: None,
768                tokens: None,
769                message: "Failed to logout".to_string(),
770            }),
771        );
772    }
773
774    let _ = log_audit_event(
775        &db,
776        AuditEvent {
777            log_type: "logout_success".to_string(),
778            user_id: Some(payload.user_id),
779            ..Default::default()
780        },
781    )
782    .await;
783
784    (
785        StatusCode::OK,
786        Json(AuthTokenResponse {
787            success: true,
788            user: None,
789            tokens: None,

795/// Handles user profile update requests
796/// This endpoint allows clients to update their user profile information,
797/// such as username and email. It verifies the provided data, updates the user profile in the database, and logs the operation.
798///
799/// # Arguments
800/// * `State(db)` - Shared state containing the database connection
801/// * `Json(payload)` - The profile update request payload containing the user ID and new profile information
802///
803/// # Returns
804/// * `(StatusCode, Json<AuthTokenResponse>)` - The HTTP status code and JSON response indicating success or failure of the profile update operation
805///
806/// # Example
807/// ```rust
808/// let payload = UpdateUserProfileRequest {
809///     user_id: "user_id".to_string(),
810///     username: Some("new_username".to_string()),
811///     email: Some("new_email@example.com".to_string()),
812/// };
813/// ```
814pub async fn update_user_profile(
815    State(db): State<DB>,
816    Json(payload): Json<UpdateUserProfileRequest>,
817) -> (StatusCode, Json<AuthTokenResponse>) {
818    if payload.username.trim().is_empty() {
819        return (
820            StatusCode::BAD_REQUEST,
821            Json(AuthTokenResponse {
822                success: false,
823                user: None,
824                tokens: None,
825                message: "Username cannot be empty".to_string(),
826            }),
827        );
828    }
829
830    let user = match auth::update_user_profile(&db, &payload).await {
831        Ok(user) => user,
832        Err((status, message)) => {
833            let _ = log_audit_event(
834                &db,
835                AuditEvent {
836                    log_type: "update_user_profile_failed".to_string(),
837                    action: Some(message.clone()),
838                    target_type_table: Some("users".to_string()),
839                    target_type_table_id: Some(payload.user_id.clone()),
840                    user_id: Some(payload.user_id),
841                    ..Default::default()
842                },
843            )
844            .await;
845
846            return (
847                status,
848                Json(AuthTokenResponse {
849                    success: false,
850                    user: None,
851                    tokens: None,
852                    message,
853                }),
854            );
855        }
856    };
857
858    let _ = log_audit_event(
859        &db,
860        AuditEvent {
861            log_type: "update_user_profile_success".to_string(),
862            target_type_table: Some("users".to_string()),
863            target_type_table_id: Some(payload.user_id.clone()),
864            user_id: Some(payload.user_id),
865            ..Default::default()
866        },
867    )
868    .await;
869
870    (
871        StatusCode::OK,
872        Json(AuthTokenResponse {
873            success: true,
874            user: Some(user.to_response()),
875            tokens: None,
876            message: "Profile updated successfully".to_string(),
877        }),
878    )
