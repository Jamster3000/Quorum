6/// Handles user signup
7///
8/// Allows anyone with access to the server to be able to signup and create an account.
9/// When this is called it prompts for username, email (optional) and password one by one, where password is typed hidden.
10///
11/// # Arguments
12/// * `db` - A reference to the database connection.
13///
14/// # Example
15/// ```rust
16/// signup(&db).await; // Prompts for username, password, and optional email to create a new user
17/// ```
18pub async fn signup(db: &DB) {
19    let username = prompt("  Username: ");
20    let password = prompt_password("  Password: ");
21    let email = prompt("  Email (optional, leave blank): ");
22
23    let email = if email.is_empty() { None } else { Some(email) };
24
25    match crate::db::queries::auth::signup_user(db, &username, email.as_deref(), &password, None)
26        .await
27    {
28        Ok(user) => println!(
29            "{}",
30            format!("  Created user: {} ({:?})", user.username, user.id.key).green()
31        ),
32        Err(e) => println!("{}", format!("  Failed to create user: {}", e).red()),
33    }
34}

36/// Handles user login
37///
38/// Prompts for username and password, and verifies the credentials against the database.
39/// When this is called it prompts for username and password one by one, where password is typed hidden.
40///
41/// # Arguments
42/// * `db` - A reference to the database connection.
43/// * `session` - A reference to the admin session, used to track login state.
44///
45/// # Example
46/// ```rust
47/// login(&db, &session).await; // Prompts for username and password to log in
48/// ```
49pub async fn login(db: &DB, session: &Arc<Mutex<AdminSession>>) {
50    {
51        let sess = session.lock().unwrap();
52        if sess.is_valid() {
53            println!(
54                "{}",
55                format!(
56                    "Already logged in as {}. Run server:logout first.",
57                    sess.username.as_deref().unwrap_or("unknown")
58                )
59                .yellow()
60            );
61            return;
62        }
63    }
64
65    let username = prompt("  Username: ");
66    let password = prompt_password("  Password: ");
67
68    match crate::db::queries::auth::verify_user_credentials(db, &username, &password).await {
69        Ok(user) => {
70            let mut sess = session.lock().unwrap();
71            sess.login(user.username.clone(), user.is_admin);
72            println!(
73                "{}",
74                format!("  Logged in as {}.", user.username).green().bold()
75            );
76            println!(
77                "{}",
78                format!(
79                    "  Session will expire after {} minutes of inactivity.",
80                    crate::cli::SESSION_TIMEOUT_MINS
81                )
82                .dimmed()
83            );
84        }
85        Err(_) => {
86            println!("{}", "  Invalid username or password.".red());
87        }
88    }
89
90    println!();
91}

93/// Handles making a user an admin
94///
95/// This uses a username as reference to mark a user as an admin.
96///
97/// # Arguments
98/// * `db` - A reference to the database connection.
99/// * `username` - The username of the user to be made an admin.
100/// * `session` - A reference to the admin session, used to update the current session if the logged-in user is made an admin.
101///
102/// # Example
103/// ```rust
104/// make_admin(&db, "some_user", &session).await;
105/// ```
106pub async fn make_admin(db: &DB, username: &str, session: &Arc<Mutex<AdminSession>>) {
107    match crate::db::queries::auth::make_admin(db, username).await {
108        Ok(_) => {
109            println!("{}", format!("  {} is now an admin.", username).green());
110
111            let mut sess = session.lock().unwrap();
112            if sess.username.as_deref() == Some(username) {
113                sess.update_is_admin(true);
114            }
115        }
116        Err(e) => println!("{}", format!("  Failed: {}", e).red()),
117    }
118}

120/// Prompts the user for input with a given label and returns the trimmed input as a String.
121///
122/// # Arguments
123/// * `label` - The label to display before the input prompt.
124///
125/// # returns
126/// A String containing the user's input, trimmed of whitespace.
127fn prompt(label: &str) -> String {
128    print!("{}", label.white());
129    std::io::Write::flush(&mut std::io::stdout()).ok();
130
131    let mut input = String::new();
132    std::io::stdin().read_line(&mut input).ok();
133    input.trim().to_string()
134}

137///
138/// # Arguments
139/// * `label` - The label to display before the password prompt.
140///

142/// A String containing the user's password input, or an empty string if reading the password fails.
143fn prompt_password(label: &str) -> String {
144    print!("{}", label.white());
145    std::io::Write::flush(&mut std::io::stdout()).ok();
146
147    rpassword::read_password().unwrap_or_default()
