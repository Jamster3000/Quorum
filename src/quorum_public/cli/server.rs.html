13pub async fn reinitialize_schema(db: &DB) -> Result<(), String> {
14    let timer = startup::create_timer();
15
16    match schema::init(db).await {
17        Ok(()) => {
18            startup::print_step("Re-initializing schema", true, startup::elapsed(timer));
19
20            Ok(())
21        }
22
23        Err(error) => {
24            startup::print_step("Re-initializing schema", false, startup::elapsed(timer));
25
26            eprintln!("{}", format!("  Error: {}", error).red());
27
28            let _ = core_db::queries::server_logs::log_error(db, error.to_string(), 0).await;
29
30            Err(error.to_string())
31        }
32    }
33}

35/// Authenticates an admin session using the server passphrase.
36///
37/// Prompts for the passphrase and attempts to decrypt secrets.enc with it.
38/// Success means the caller is whoever set up the server — session is granted.
39/// The session expires after 20 minutes of inactivity as normal.
40///
41/// # Arguments
42/// * `session` - A reference to the admin session, used to track login state.
43///
44/// # Example
45/// ```rust
46/// login(&session);
47/// ```
48pub fn login(session: &Arc<Mutex<AdminSession>>) {
49    {
50        let sess = session.lock().unwrap();
51        if sess.is_valid() {
52            println!("{}", "Already logged in. Run server:logout first.".yellow());
53            return;
54        }
55    }
56
57    let passphrase = match quorum_core::utility::secrets::prompt_passphrase() {
58        Ok(p) => p,
59        Err(e) => {
60            println!("{}", format!("  Failed to read passphrase: {}", e).red());
61            return;
62        }
63    };
64
65    match quorum_core::utility::secrets::load_encrypted_config(&passphrase) {
66        Ok(_) => {
67            let mut sess = session.lock().unwrap();
68            sess.login("admin".to_string(), true);
69            println!("{}", "  Authenticated.".green().bold());
70            println!(
71                "{}",
72                format!(
73                    "  Session will expire after {} minutes of inactivity.",
74                    crate::cli::SESSION_TIMEOUT_MINS
75                )
76                .dimmed()
77            );
78        }
79        Err(_) => {
80            println!("{}", "  Invalid passphrase.".red());
81        }
82    }
83
84    println!();
85}

87/// Prompts the user to confirm database deletion and initiates a reset.
88///
89/// Displays a confirmation prompt and requires the user to enter their server passphrase
90/// to verify they have authority to perform this destructive operation. On confirmation,
91/// deletes the entire database directory and recreates it empty, then gracefully shuts
92/// down the server for restart.
93///
94/// The user will be prompted twice: once to confirm intent, and once to provide the passphrase.
95/// Either prompt can be cancelled by answering "no" to the confirmation or entering an invalid passphrase.
96///
97/// # Arguments
98/// * `shutdown_tx` - A watch channel sender used to signal server shutdown after a successful reset.
99///
100/// # Example
101/// ```rust
102/// confirm_and_delete(&shutdown_tx).await;
103/// ```
104pub async fn confirm_and_delete(shutdown_tx: &tokio::sync::watch::Sender<bool>) {
105    println!();
106    let _ = typewriter_println(&format!(
107        "{}",
108        "WARNING: This will delete all data and shut down the server.\nYou must restart it manually after.\n\nContinue? (y/n)".yellow()
109    ))
110    .map_err(|e| e.to_string());
111
112    let mut confirm = Input::<String>::new()
113        .with_prompt(" ")
114        .interact_text()
115        .unwrap();
116
117    confirm = confirm.to_lowercase();
118
119    if confirm != "y" && confirm != "yes" {
120        return;
121    }
122
123    println!();
124    let _ = typewriter_println(&format!(
125        "{}",
126        "Enter passphrase to confirm database deletion..."
127            .cyan()
128            .bold()
129    ))
130    .map_err(|e| e.to_string());
131
132    match prompt_passphrase() {
133        Ok(passphrase) => {
134            let passphrase = Zeroizing::new(passphrase);
135            if let Err(e) = perform_reset(&passphrase).await {
136                eprintln!("{}", format!("Failed to reset database: {e}").red());
137                return;
138            }
139            println!(
140                "{}",
141                "Database reset complete. The server is shutting down.\nPlease restart the server to continue.".green()
142            );
143            let _ = shutdown_tx.send(true);
144        }
145        Err(e) => {
146            eprintln!("Failed to read passphrase: {}", e);
147        }
148    }
149}

151/// Deletes and reinitializes the database with an empty schema.
152///
153/// Verifies the provided passphrase to ensure the caller has authority, then spawns a blocking
154/// task to delete the entire database directory and recreate it. After the filesystem operations
155/// complete, reinitializes the database schema from the initial migration script.
156///
157/// The database deletion is performed on a separate thread to avoid holding locks from the CLI's
158/// database reference, which would prevent the filesystem operations from succeeding on Windows.
159///
160/// # Errors
161///
162/// Returns an error when:
163/// - the passphrase is invalid,
164/// - the database directory cannot be deleted or recreated,
165/// - or schema initialization fails.
166///
167/// # Arguments
168/// * `passphrase` - The server passphrase, used to verify the caller's authority before performing the reset.
169async fn perform_reset(passphrase: &str) -> Result<(), Box<dyn std::error::Error>> {
170    quorum_core::utility::config::Config::load_with_passphrase(passphrase)?;
171
172    let config = quorum_core::utility::config::Config::get();
173    let path = std::path::Path::new(&config.surreal_data_path).to_path_buf();
174
175    tokio::task::spawn_blocking(move || {
176        std::thread::sleep(std::time::Duration::from_millis(1000));
177        if path.exists() {
178            std::fs::remove_dir_all(&path).ok();
179        }
180        std::fs::create_dir_all(&path).ok();
181    })
182    .await?;
183
184    Ok(())
185}
