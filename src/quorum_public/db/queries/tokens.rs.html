8/// Stores a new refresh token for a user in the database.
9/// The token is hashed before storage for security purpsoes.
10///
11/// # Arguments
12/// * `db` - A reference to the database connection.
13/// * `user_id` - The ID of the user the token belongs to.
14/// * `refresh_token` - The refresh token to be stored.
15/// * `expires_at` - The expiration time of the refresh token (Unix timestamp).
16///
17/// # Returns
18/// * `Ok(())` if the token was successfully stored.
19/// * `Err` if there was an error during the database operation.
20///
21/// # Exmaple
22/// ```rust
23/// use crate::db::DB;
24/// use crate::db::queries::tokens::store_refresh_token;
25/// let db = DB::new("localhost:8529", "my_database").await.unwrap();
26/// store_refresh_token(&db, "user123", "some_refresh_token", 1700000000).await.unwrap();
27/// ```
28pub async fn store_refresh_token(
29    db: &DB,
30    user_id: &str,
31    refresh_token: &str,
32    expires_at: i64,
33) -> Result<(), Box<dyn Error + Send + Sync>> {
34    let token_hash = hash_token(refresh_token);
35
36    let query = "CREATE refresh_token SET user_id = $user_id, token_hash = $token_hash, expires_at = $expires_at";
37
38    db.query(query)
39        .bind(("user_id", format!("users:{}", user_id)))
40        .bind(("token_hash", token_hash))
41        .bind(("expires_at", expires_at))
42        .await?;
43
44    Ok(())
45}

47/// Revokes a refresh token by setting its `is_revoked` flag to true in the database.
48///
49/// # Arguments
50/// * `db` - A reference to the database connection.
51/// * `refresh_token` - The refresh token to be revoked.
52///
53/// # Returns
54/// * `Ok(())` if the token was successfully revoked.
55/// * `Err` if there was an error during the database operation.
56///
57/// # Example
58/// ```rust
59/// use crate::db::DB;
60/// use crate::db::queries::tokens::revoke_refresh_token;
61/// let db = DB::new("localhost:8529", "my_database").await.unwrap();
62/// revoke_refresh_token(&db, "some_refresh_token").await.unwrap();
63/// ```
64pub async fn revoke_refresh_token(
65    db: &DB,
66    refresh_token: &str,
67) -> Result<(), Box<dyn Error + Send + Sync>> {
68    let token_hash = hash_token(refresh_token);
69    let query = "UPDATE refresh_token SET is_revoked = true WHERE token_hash = $token_hash";
70
71    db.query(query).bind(("token_hash", token_hash)).await?;
72
73    Ok(())
74}

76/// Deletes all refresh tokens associated with a user from the database.
77///
78/// # Arguments
79/// * `db` - A reference to the database connection.
80/// * `user_id` - The ID of the user whose tokens should be deleted.
81///
82/// # Returns
83/// * `Ok(())` if the tokens were successfully deleted.
84/// * `Err` if there was an error during the database operation.
85///
86/// # Example
87/// ```rust
88/// use crate::db::DB;
89/// use crate::db::queries::tokens::delete_all_user_tokens;
90/// let db = DB::new("localhost:8529", "my_database").await.unwrap();
91/// delete_all_user_tokens(&db, "user123").await.unwrap();
92/// ```
93pub async fn delete_all_user_tokens(
94    db: &DB,
95    user_id: &str,
96) -> Result<(), Box<dyn Error + Send + Sync>> {
97    let query = "DELETE FROM refresh_token WHERE user_id = $user_id";
98
99    db.query(query)
100        .bind(("user_id", format!("users:{}", user_id)))
101        .await?;
102
103    Ok(())
104}

106/// Hashes a refresh token using SHA-256 for secure storage in the database.
107///
108/// # Arguments
109/// * `token` - The refresh token to be hashed.
110///
111/// # Returns
112/// * A `String` representing the SHA-256 hash of the token.
113///
114/// # Example
115/// ```rust
116/// let token = "some_refresh_token";
117/// let hashed_token = hash_token(token);
118/// ```
119fn hash_token(token: &str) -> String {
120    let mut hasher = Sha256::new();
121    hasher.update(token.as_bytes());
122    hex::encode(hasher.finalize())
123}
