1//! Centralized configuration management.
2//!
3//! Configuration is loaded once at startup from an AES-256-GCM encrypted file (`secrets.enc`),
4//! decrypted using a passphrase the admin enters at the terminal. No `.env` files, no plaintext
5//! secrets on disk.
6//!
7//! The loaded config is stored in a global [`ArcSwap`] singleton, meaning:
8//! - All reads are atomic and lock-free via [`Config::get()`]
9//! - The config can be hot-reloaded at runtime via [`Config::reload()`] without restarting the server
10//! - Any reload is immediately visible to all subsequent [`Config::get()`] calls across all threads
11
12use crate::utility::secrets::{

14};
15use crate::utility::std::{press_enter_to_continue, typewriter_println};
16use arc_swap::ArcSwap;
17use colored::Colorize;

21/// The global configuration singleton.
22///
23/// Populated once by [`Config::load()`] and never replaced — only the inner [`Arc<Config>`]
24/// is swapped on reload, which is what [`ArcSwap`] is for.
25static CONFIG: OnceLock<ArcSwap<Config>> = OnceLock::new();
26
27/// Server-wide configuration, loaded from the encrypted `secrets.enc` file at startup.
28///
29/// All fields are read-only after initialization. To update values at runtime,
30/// edit `secrets.enc` and call [`Config::reload()`].
31#[derive(Debug)]
32pub struct Config {
33    /// HTTP port the server binds to (e.g. `3000`).
34    pub server_port: u16,
35
36    /// Full server URL including protocol and port (e.g. `http://127.0.0.1:3000`).
37    /// Used by the test suite and any internal self-referencing requests.
38    pub server_url: String,
39
40    /// IP address the server binds to (e.g. `127.0.0.1` or `0.0.0.0`).
41    pub server_host: String,
42
43    /// Path to the directory where SurrealDB stores its on-disk data.
44    ///
45    /// SurrealDB runs embedded (in-process) using RocksDB as the storage engine.
46    /// This directory is created automatically on first startup if it doesn't exist.
47    /// Defaults to `./data/db` if not set during setup.
48    pub surreal_data_path: String,
49
50    /// SurrealDB namespace to use (e.g. `quorum`).
51    pub surreal_ns: String,
52
53    /// SurrealDB database name within the namespace (e.g. `quorum`).
54    pub surreal_db: String,
55
56    /// Secret key used to sign and verify JWT access and refresh tokens.
57    ///
58    /// Auto-generated as a random 32-byte hex string during first-run setup.
59    /// Changing this value invalidates all currently issued tokens.
60    pub jwt_secret: String,
61
62    /// How long an access token remains valid, in minutes.
63    /// Access tokens are short-lived by design — default is 15 minutes.
64    pub jwt_access_minutes: i64,
65
66    /// How long a refresh token remains valid, in days.
67    /// Refresh tokens are long-lived — default is 7 days.
68    pub jwt_refresh_days: i64,
69
70    /// Whether to run the functional and robustness test suite on server startup.
71    ///
72    /// Should be `false` in production. When `true`, the test suite fires real HTTP
73    /// requests against the running server immediately after it becomes ready.
74    pub enable_testing: bool,
75
76    /// Rate limit: sustained request rate for standard endpoints (requests per second).
77    pub default_per_second: u64,
78
79    /// Rate limit: maximum burst size for standard endpoints.
80    ///
81    /// Allows short spikes above `default_per_second` up to this many requests
82    /// before the limiter kicks in.
83    pub default_burst_size: u32,
84
85    /// Rate limit: sustained request rate for test/dev endpoints (requests per second).
86    ///
87    /// Intentionally higher than `default_per_second` so the test suite doesn't
88    /// rate-limit itself during a full run.
89    pub testing_per_second: u64,
90
91    /// Rate limit: maximum burst size for test/dev endpoints.
92    pub testing_burst_size: u32,
93}

95impl Config {
96    /// Builds a [`Config`] directly from a decrypted [`SerializableConfig`].
97    ///
98    /// This is the only construction path — there are no environment variable fallbacks.
99    /// All values come from the encrypted secrets file, which was populated during first-run setup.
100    ///
101    /// [`SerializableConfig`]: crate::utility::secrets::SerializableConfig
102    fn from_serializable(s: crate::utility::secrets::SerializableConfig) -> Config {
103        Config {
104            server_url: format!("http://{}:{}", s.server_host, s.server_port),
105            server_port: s.server_port,
106            server_host: s.server_host,
107            surreal_data_path: s.surreal_data_path,
108            surreal_ns: s.surreal_ns,
109            surreal_db: s.surreal_db,
110            jwt_secret: s.jwt_secret,
111            jwt_access_minutes: s.jwt_access_minutes,
112            jwt_refresh_days: s.jwt_refresh_days,
113            enable_testing: s.enable_testing,
114            default_per_second: s.default_per_second,
115            default_burst_size: s.default_burst_size,
116            testing_per_second: s.testing_per_second,
117            testing_burst_size: s.testing_burst_size,
118        }
119    }
120
121    /// Loads configuration from `secrets.enc` and initializes the global [`CONFIG`] singleton.
122    ///
123    /// On first run (no `secrets.enc` exists), walks the admin through the interactive setup
124    /// wizard, prompts for a passphrase, encrypts the config, saves it, then loads it.
125    ///
126    /// On subsequent runs, prompts for the passphrase and decrypts the existing file.
127    ///
128    /// This must be called exactly once, before any call to [`Config::get()`].
129    /// Calling it a second time returns an error (`"Config already initialized"`).
130    ///
131    /// # Errors
132    /// * Wrong passphrase — AES-GCM auth tag mismatch, decryption fails
133    /// * Corrupt or missing `secrets.enc` — file read or deserialization fails
134    /// * Setup wizard aborted — terminal I/O error during first-run prompts
135    ///
136    /// # Example
137    /// ```rust
138    /// Config::load().expect("Failed to load configuration");
139    /// ```
140    pub fn load() -> Result<(), Box<dyn std::error::Error>> {
141        if secrets_exist() {
142            println!();
143            typewriter_println(&format!(
144                "{}",
145                "Enter passphrase to unlock the server...".cyan().bold()
146            ))
147            .map_err(|e| e.to_string())?;
148
149            let passphrase = prompt_passphrase()?;
150            Self::load_with_passphrase(&passphrase)
151        } else {
152            let serializable = run_setup()?;
153            let passphrase = prompt_passphrase()?;
154
155            println!();
156            typewriter_println(&format!(
157                "{}",
158                "Passphrase setup successfully!".cyan().bold()
159            ))
160            .map_err(|e| e.to_string())?;
161
162            press_enter_to_continue(true, true);
163
164            save_encrypted_config(&serializable, &passphrase)?;
165
166            Self::load()
167        }
168    }
169
170    /// Reloads configuration from `secrets.enc` without restarting the server.
171    ///
172    /// Prompts the admin for their passphrase, decrypts the file, and atomically swaps
173    /// the global config. All subsequent [`Config::get()`] calls will see the new values
174    /// immediately, with no downtime and no impact on in-flight requests.
175    ///
176    /// Useful when you've updated `secrets.enc` (e.g. changed rate limits or JWT expiry)
177    /// and want the changes applied without a full server restart.
178    ///
179    /// # Errors
180    /// * Wrong passphrase — decryption fails
181    /// * Corrupt `secrets.enc` — deserialization fails
182    ///
183    /// # Panics
184    /// Panics if called before [`Config::load()`].
185    pub fn reload() -> Result<(), Box<dyn std::error::Error>> {
186        typewriter_println(&format!(
187            "{}",
188            "Enter passphrase to reload config...".cyan().bold()
189        ))
190        .map_err(|e| e.to_string())?;
191
192        let passphrase = prompt_passphrase()?;
193        let serializable = load_encrypted_config(&passphrase)?;
194        let config = Self::from_serializable(serializable);
195
196        CONFIG
197            .get()
198            .expect("Config not initialized. Call Config::load() first.")
199            .store(Arc::new(config));
200
201        Ok(())
202    }
203
204    pub fn load_with_passphrase(passphrase: &str) -> Result<(), Box<dyn std::error::Error>> {
205        let serializable = load_encrypted_config(passphrase)
206            .map_err(|e| -> Box<dyn std::error::Error> { e.into() })?;
207        let config = Self::from_serializable(serializable);
208
209        if let Some(existing) = CONFIG.get() {
210            existing.store(Arc::new(config));
211            Ok(())
212        } else {
213            CONFIG
214                .set(ArcSwap::from_pointee(config))
215                .map_err(|_| "Config already initialized".into())
216        }
217    }
218
219    /// Returns a snapshot of the current configuration.
220    ///
221    /// The returned guard is cheap to acquire (atomic load, no locking) and reflects
222    /// the latest [`Config::reload()`] if one has been called. Treat it like an `Arc<Config>` —
223    /// hold it for the duration of a request or operation, then drop it.
224    ///
225    /// # Panics
226    /// Panics if [`Config::load()`] has not been called yet.
227    ///
228    /// # Example
229    /// ```rust
230    /// let config = Config::get();
231    /// println!("Listening on port {}", config.server_port);
232    /// ```
233    pub fn get() -> arc_swap::Guard<Arc<Config>> {
234        CONFIG
235            .get()
236            .expect("Config not initialized. Call Config::load() first.")
237            .load()
238    }
239}
