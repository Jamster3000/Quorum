1//! Centralized configuration management.
2//!
3//! This module loads all environment variables once at application startup and provides
4//! a singleton `Config` struct that can be accessed from anywhere in the application.
5//! This eliminates the need to repeatedly read `.env` files and ensures consistent configuration
6//! values across the entire application.
7//!
8//! Unlike a `OnceLock`, the config can be reloaded at runtime via `Config::reload()` without
9//! restarting the server. All reads are atomic and lock-free via `arc-swap`.
10
11use crate::utility::secrets::{
12    load_encrypted_config, prompt_passphrase, run_setup, save_encrypted_config, secrets_exist,
13};
14use crate::utility::std::{press_enter_to_continue, typewriter_println};
15use arc_swap::ArcSwap;
16use colored::Colorize;
17use std::sync::Arc;
18use std::sync::OnceLock;
19
20#[derive(Debug)]
21pub struct Config {
22    /// HTTP server port number.
23    pub server_port: u16,
24
25    /// Full server URL including protocol and port (e.g., `http://127.0.0.1:3000`).
26    /// Used by clients and tests to connect to the server.
27    pub server_url: String,
28
29    /// Server host/IP address to bind to.
30    pub server_host: String,
31
32    /// SurrealDB connection URL.
33    pub surreal_url: String,
34
35    /// SurrealDB root username for authentication.
36    pub surreal_user: String,
37
38    /// SurrealDB root password for authentication.
39    pub surreal_pass: String,
40
41    /// SurrealDB namespace name.
42    pub surreal_ns: String,
43
44    /// SurrealDB database name.
45    pub surreal_db: String,
46
47    /// Secret key for signing and verifying JWT tokens.
48    pub jwt_secret: String,
49
50    /// Access token expiry time in minutes.
51    pub jwt_access_minutes: i64,
52
53    /// Refresh token expiry time in days.
54    pub jwt_refresh_days: i64,
55
56    /// Whether to run functional tests on server startup.
57    pub enable_testing: bool,
58
59    /// Rate limit: requests per second for default endpoints.
60    pub default_per_second: u64,
61
62    /// Rate limit: burst size for default endpoints.
63    pub default_burst_size: u32,
64
65    /// Rate limit: requests per second for testing endpoints.
66    pub testing_per_second: u64,
67
68    /// Rate limit: burst size for testing endpoints.
69    pub testing_burst_size: u32,
70}
71
72static CONFIG: OnceLock<ArcSwap<Config>> = OnceLock::new();
73
74impl Config {
75    /// Parses all environment variables into a `Config` instance.
76    fn build(
77        jwt_secret: String,
78        surreal_pass: String,
79    ) -> Result<Config, Box<dyn std::error::Error>> {
80        let server_port: u16 = std::env::var("SERVER_PORT")
81            .unwrap_or_else(|_| "3000".to_string())
82            .parse()?;
83
84        let server_host = std::env::var("SERVER_HOST").unwrap_or_else(|_| "127.0.0.1".to_string());
85
86        Ok(Config {
87            server_port,
88            server_host: server_host.clone(),
89            server_url: std::env::var("SERVER_URL")
90                .unwrap_or_else(|_| format!("http://{}:{}", server_host, server_port)),
91            surreal_url: std::env::var("SURREAL_URL")?,
92            surreal_user: std::env::var("SURREAL_USER")?,
93            surreal_pass,
94            surreal_ns: std::env::var("SURREAL_NS")?,
95            surreal_db: std::env::var("SURREAL_DB")?,
96            jwt_secret,
97            jwt_access_minutes: std::env::var("JWT_ACCESS_MINUTES")?.parse()?,
98            jwt_refresh_days: std::env::var("JWT_REFRESH_DAYS")?.parse()?,
99            enable_testing: std::env::var("ENABLE_TESTING")
100                .unwrap_or_else(|_| "true".to_string())
101                .parse()?,
102            default_per_second: std::env::var("DEFAULT_PER_SECOND")
103                .unwrap_or_else(|_| "2".to_string())
104                .parse()?,
105            default_burst_size: std::env::var("DEFAULT_BURST_SIZE")
106                .unwrap_or_else(|_| "5".to_string())
107                .parse()?,
108            testing_per_second: std::env::var("TESTING_PER_SECOND")
109                .unwrap_or_else(|_| "10".to_string())
110                .parse()?,
111            testing_burst_size: std::env::var("TESTING_BURST_SIZE")
112                .unwrap_or_else(|_| "50".to_string())
113                .parse()?,
114        })
115    }
116
117    /// Loads configuration from environment variables and initializes the global `CONFIG` singleton.
118    ///
119    /// # Returns
120    /// * `Ok(())` if the configuration was loaded successfully.
121    /// * `Err` if there was an error loading the configuration, such as missing or invalid environment variables.
122    ///
123    /// Example
124    /// ```rust
125    /// Config::load().expect("Failed to load configuration");
126    /// ```
127    pub fn load() -> Result<(), Box<dyn std::error::Error>> {
128        if secrets_exist() {
129            println!();
130            typewriter_println(&format!(
131                "{}",
132                "Enter passphrase to unlock the server...".cyan().bold()
133            ))
134            .map_err(|e| e.to_string())?;
135
136            let passphrase = prompt_passphrase()?;
137            let serializable_config = load_encrypted_config(&passphrase)?;
138
139            let config = Config {
140                server_port: serializable_config.server_port,
141                server_host: serializable_config.server_host.clone(),
142                server_url: format!(
143                    "http://{}:{}",
144                    serializable_config.server_host, serializable_config.server_port
145                ),
146                surreal_url: serializable_config.surreal_url,
147                surreal_user: serializable_config.surreal_user,
148                surreal_pass: serializable_config.surreal_pass,
149                surreal_ns: serializable_config.surreal_ns,
150                surreal_db: serializable_config.surreal_db,
151                jwt_secret: serializable_config.jwt_secret,
152                jwt_access_minutes: serializable_config.jwt_access_minutes,
153                jwt_refresh_days: serializable_config.jwt_refresh_days,
154                enable_testing: serializable_config.enable_testing,
155                default_per_second: serializable_config.default_per_second,
156                default_burst_size: serializable_config.default_burst_size,
157                testing_per_second: serializable_config.testing_per_second,
158                testing_burst_size: serializable_config.testing_burst_size,
159            };
160            CONFIG
161                .set(ArcSwap::from_pointee(config))
162                .map_err(|_| "Config already initialized".into())
163        } else {
164            let serializable_config = run_setup()?;
165            let passphrase = prompt_passphrase()?;
166
167            println!();
168            typewriter_println(&format!(
169                "{}",
170                "Passphrase setup successfully!".cyan().bold()
171            ))
172            .map_err(|e| e.to_string())?;
173
174            press_enter_to_continue(true, true);
175
176            save_encrypted_config(&serializable_config, &passphrase)?;
177            Self::load()
178        }
179    }
180
181    /// Reloads configuration from environment variables without restarting the server.
182    /// All subsequent `Config::get()` calls will see the new values atomically.
183    ///
184    /// # Errors
185    /// * Same as `Config::load()` — missing or invalid environment variables
186    /// * Panics if called before `Config::load()`
187    pub fn reload() -> Result<(), Box<dyn std::error::Error>> {
188        let current = Self::get();
189        let jwt_secret = current.jwt_secret.clone();
190        let surreal_pass = current.surreal_pass.clone();
191        drop(current);
192
193        let config = Self::build(jwt_secret, surreal_pass)?;
194        CONFIG
195            .get()
196            .expect("Config not initialized. Call Config::load() first.")
197            .store(Arc::new(config));
198        Ok(())
199    }
200
201    /// Returns a snapshot of the current configuration.
202    /// The snapshot is cheap to load and reflects any reloads since the last call.
203    ///
204    /// # Panics
205    /// Panics if `Config::load()` has not been called yet.
206    pub fn get() -> arc_swap::Guard<Arc<Config>> {
207        CONFIG
208            .get()
209            .expect("Config not initialized. Call Config::load() first.")
210            .load()
211    }
212}
