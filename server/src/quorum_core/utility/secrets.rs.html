28fn generate_random_bytes() -> [u8; 32] {
29    let mut bytes = [0u8; 32];
30    OsRng.fill_bytes(&mut bytes);
31    bytes
32}

55/// Derives a key from the given passphrase and salt using Argon2id.
56///
57/// # Arguments
58/// * `passphrase` - The passphrase to derive the key from.
59/// * `salt` - A 32-byte salt used in the key derivation process.
60///
61/// # Returns
62/// A `Zeroizing<[u8; 32]>` containing the derived key. The
63fn derive_key(passphrase: &str, salt: &[u8; 32]) -> Zeroizing<[u8; 32]> {
64    use argon2::{Algorithm, Params, Version};
65    let argon2 = argon2::Argon2::new(
66        Algorithm::Argon2id,
67        Version::V0x13,
68        Params::new(65536, 3, 1, Some(32)).unwrap(),
69    );
70    let mut key = Zeroizing::new([0u8; 32]);
71    argon2
72        .hash_password_into(passphrase.as_bytes(), salt, &mut *key)
73        .expect("Failed to derive key");
74    key
75}

77/// Encrypts the given data using AES-256-GCM with a key derived from the provided passphrase.
78///
79/// # Arguments
80/// * `data` - The plaintext data to encrypt.
81/// * `passphrase` - The passphrase used to derive the encryption key.
82///
83/// # Returns
84/// A `Result` containing the encrypted data as a `Vec<u8>` on success, or an error message as a `String` on failure.
85fn encrypt(data: &[u8], passphrase: &str) -> Result<Vec<u8>, String> {
86    let salt = generate_random_bytes();
87    let key = derive_key(passphrase, &salt);
88    let cipher = Aes256Gcm::new_from_slice(&*key).unwrap();
89    let nonce = Aes256Gcm::generate_nonce(&mut OsRng);
90    cipher
91        .encrypt(&nonce, data)
92        .map(|mut ciphertext| {
93            let mut result = Vec::with_capacity(32 + 12 + ciphertext.len());
94            result.extend_from_slice(&salt);
95            result.extend_from_slice(nonce.as_slice());
96            result.append(&mut ciphertext);
97            result
98        })
99        .map_err(|e| format!("Encryption failed: {}", e))
100}

102/// Decrypts the given encrypted data using AES-256-GCM with a key derived from the provided passphrase.
103///
104/// # Arguments
105/// * `encrypted_data` - The encrypted data to decrypt, which should include the salt and nonce.
106/// * `passphrase` - The passphrase used to derive the decryption key.
107///
108/// # Returns
109/// A `Result` containing the decrypted data as a `Vec<u8>` on success, or an error message as a `String` on failure.
110fn decrypt(encrypted_data: &[u8], passphrase: &str) -> Result<Vec<u8>, String> {
111    if encrypted_data.len() < 44 {
112        return Err("Encrypted data too short".to_string());
113    }
114    let (salt_bytes, rest) = encrypted_data.split_at(32);
115    let (nonce_bytes, ciphertext) = rest.split_at(12);
116    let salt: [u8; 32] = salt_bytes.try_into().map_err(|_| "Invalid salt length")?;
117    let nonce = Nonce::from_slice(nonce_bytes);
118    let key = derive_key(passphrase, &salt);
119    let cipher = Aes256Gcm::new_from_slice(&*key).unwrap();
120    cipher
121        .decrypt(nonce, ciphertext)
122        .map_err(|e| format!("Decryption failed: {}", e))
123}

125/// Saves the given configuration to an encrypted file using the provided passphrase.
126///
127/// Encrypts the config and writes it to both the primary file (`secrets.enc`) and a backup
128/// (`backups/secrets.enc.backup`). The backup directory is created automatically if it doesn't exist.
129/// The backup is written first to ensure a valid copy exists even if the primary write fails.
130///
131/// # Arguments
132/// * `config` - The configuration to save.
133/// * `passphrase` - The passphrase used to encrypt the configuration.
134///
135/// # Returns
136/// A `Result` indicating success or failure, with an error message as a `String` on failure.
137///
138/// # Errors
139/// Returns an error if:
140/// - Serialization of the config fails
141/// - Encryption fails
142/// - The backups directory cannot be created
143/// - Either the backup or primary file cannot be written
144///
145/// # Example
146/// ```rust
147/// let config = SerializableConfig { ... };
148/// save_encrypted_config(&config, "my_secure_passphrase").expect("Failed to save encrypted config");
149/// ```
150pub fn save_encrypted_config(config: &SerializableConfig, passphrase: &str) -> Result<(), String> {
151    let json = serde_json::to_vec(config).map_err(|e| format!("Serialization failed: {}", e))?;
152    let encrypted = encrypt(&json, passphrase)?;
153
154    // Calculate checksum
155    let mut hasher = sha2::Sha256::new();
156    hasher.update(&encrypted);
157    let checksum = hex::encode(hasher.finalize());
158
159    // Ensure backup directory exists
160    std::fs::create_dir_all("backups")
161        .map_err(|e| format!("Failed to create backups directory: {}", e))?;
162
163    // Write backup secrets file
164    let mut backup =
165        File::create(SECRETS_BACKUP_PATH).map_err(|e| format!("Failed to create backup: {}", e))?;
166    backup
167        .write_all(&encrypted)
168        .map_err(|e| format!("Failed to write backup: {}", e))?;
169
170    // Write checksum
171    fs::write(SECRETS_CHECKSUM_PATH, &checksum)
172        .map_err(|e| format!("Failed to write checksum: {}", e))?;
173
174    // Write primary secrets file
175    let mut file =
176        File::create(SECRETS_PATH).map_err(|e| format!("Failed to create secrets.enc: {}", e))?;
177    file.write_all(&encrypted)
178        .map_err(|e| format!("Failed to write secrets.enc: {}", e))?;
179
180    Ok(())
181}

183/// Loads and decrypts the configuration from the encrypted file using the provided passphrase.
184///
185/// # Arguments
186/// * `passphrase` - The passphrase used to decrypt the configuration.
187///
188/// # Returns
189/// A `Result` containing the decrypted configuration as a `SerializableConfig` on success, or an error message as a `String` on failure.
190///
191/// # Example
192/// ```rust
193/// let config = load_encrypted_config("my_secure_passphrase").expect("Failed to load encrypted config");
194/// ```
195pub fn load_encrypted_config(passphrase: &str) -> Result<SerializableConfig, String> {
196    let data = fs::read(SECRETS_PATH)
197        .or_else(|_| fs::read(SECRETS_BACKUP_PATH))
198        .map_err(|_| "secrets.enc not found.".to_string())?;
199
200    // Verify checksum if it exists
201    if let Ok(stored_checksum) = fs::read_to_string(SECRETS_CHECKSUM_PATH) {
202        let mut hasher = sha2::Sha256::new();
203        hasher.update(&data);
204        let computed = hex::encode(hasher.finalize());
205
206        if computed != stored_checksum.trim() {
207            // Primary is corrupted, try backup
208            let backup_data = fs::read(SECRETS_BACKUP_PATH)
209                .map_err(|_| "Primary file corrupted and backup missing.".to_string())?;
210            let decrypted = decrypt(&backup_data, passphrase)?;
211            return serde_json::from_slice(&decrypted)
212                .map_err(|e| format!("Failed to deserialize config: {}", e));
213        }
214    }
215
216    let decrypted = decrypt(&data, passphrase)?;
217    serde_json::from_slice(&decrypted).map_err(|e| format!("Failed to deserialize config: {}", e))
218}

320pub fn prompt_passphrase_new() -> Result<String, String> {
321    loop {
322        let passphrase = Password::new()
323            .with_prompt("Enter server passphrase")
324            .with_confirmation("Confirm passphrase", "Passphrases do not match")
325            .interact()
326            .map_err(map_dialoguer_error)?;

330
331        match validate_passphrase(&passphrase) {

338pub fn verify_admin_credentials(
339    username: &str,
340    password: &str,
341    stored_username: &str,
342    stored_hash: &str,
343) -> Result<(), String> {
344    use argon2::{
345        Algorithm, Argon2, Params, Version,
346        password_hash::{PasswordHash, PasswordVerifier},
347    };
348
349    if username != stored_username {
350        return Err("Invalid credentials".to_string());
351    }
352
353    let params = Params::new(65536, 3, 1, Some(32)).map_err(|e| e.to_string())?;
354    let argon2 = Argon2::new(Algorithm::Argon2id, Version::V0x13, params);
355
356    let parsed_hash = PasswordHash::new(stored_hash)
357        .map_err(|e| format!("Failed to parse stored hash: {}", e))?;
358
359    argon2
360        .verify_password(password.as_bytes(), &parsed_hash)
361        .map_err(|_| "Invalid credentials".to_string())
362}

364pub fn validate_passphrase(passphrase: &str) -> Result<(), String> {
365    let len = passphrase.chars().count();
366
367    if len < 12 {
368        return Err("Use at least 12 characters.".into());
369    }
370
371    if len > 64 {
372        return Err("Passphrase is too long for this field.".into());
373    }
374
375    let estimate = zxcvbn(passphrase, &[]);
376
377    if estimate.score() < Score::Three {
378        if let Some(feedback) = estimate.feedback() {
379            let warning = feedback
380                .warning()
381                .map(|w| w.to_string())
382                .unwrap_or_default();
383
384            let suggestions = feedback
385                .suggestions()
386                .iter()
387                .map(|s| s.to_string())
388                .collect::<Vec<_>>()
389                .join(" ");
390
391            let message = format!("{} {}", warning, suggestions).trim().to_string();
392
393            if message.is_empty() {
394                return Err("Passphrase is too weak.".into());
395            }
396
397            return Err(format!("Passphrase is too weak. {message}"));
398        }
399
400        return Err("Passphrase is too weak.".into());
401    }
402
403    Ok(())
